<!-- MOBILE CONTROLS HUD -->
<div id="mobileHudRoot" style="position: fixed; inset: 0; pointer-events: none; z-index: 9000; display: none;">
    <div id="mobileModePill" style="position: fixed; top: 10px; right: 10px; pointer-events: auto; background: rgba(12, 12, 20, 0.45); color: rgba(255, 255, 255, 0.7); border: 1px solid rgba(255, 255, 255, 0.18); border-radius: 14px; padding: 4px 10px; font-size: 11px; font-weight: bold; touch-action: none;">TOUCH: OFF</div>
    
    <!-- OFFICE LAYER (Only Mask) -->
    <div id="layerOffice" class="touch-layer" style="display: none; position: fixed; inset: 0;">
        <button class="t-btn" data-key="Space" data-code="32" style="position: fixed; right: 20px; bottom: 20px; width: 60px; height: 60px; pointer-events: auto; background: rgba(20, 20, 30, 0.35); color: #fff; border: 1px solid #fff3; border-radius: 12px; font-size: 12px; font-weight: bold; touch-action: none;">MASK</button>
    </div>

    <!-- SALVAGE LAYER (Joystick + Sprint/Sneak) -->
    <div id="layerSalvage" class="touch-layer" style="display: none; position: fixed; inset: 0;">
        <div id="joyZone" style="position: fixed; left: 0; top: 0; width: 50vw; height: 100vh; pointer-events: auto; touch-action: none;"></div>
        <div id="joyBase" style="position: fixed; width: 84px; height: 84px; margin-left: -42px; margin-top: -42px; border-radius: 50%; border: 1.5px solid rgba(255, 255, 255, 0.28); background: rgba(255, 255, 255, 0.06); display: none; pointer-events: none;">
            <div id="joyKnob" style="position: absolute; left: 50%; top: 50%; width: 34px; height: 34px; margin-left: -17px; margin-top: -17px; border-radius: 50%; background: rgba(255, 255, 255, 0.35);"></div>
        </div>
        <button class="t-btn" data-key="ShiftLeft" data-code="16" style="position: fixed; right: 90px; bottom: 20px; width: 60px; height: 60px; pointer-events: auto; background: rgba(20, 20, 30, 0.35); color: #fff; border: 1px solid #fff3; border-radius: 12px; font-size: 12px; font-weight: bold; touch-action: none;">SNEAK</button>
        <button class="t-btn" data-key="Space" data-code="32" style="position: fixed; right: 20px; bottom: 20px; width: 60px; height: 60px; pointer-events: auto; background: rgba(20, 20, 30, 0.35); color: #fff; border: 1px solid #fff3; border-radius: 12px; font-size: 12px; font-weight: bold; touch-action: none;">SPACE</button>
        <button class="t-btn" data-key="KeyW" data-code="87" style="position: fixed; right: 20px; bottom: 90px; width: 60px; height: 60px; pointer-events: auto; background: rgba(20, 20, 30, 0.35); color: #fff; border: 1px solid #fff3; border-radius: 12px; font-size: 12px; font-weight: bold; touch-action: none;">RUN</button>
    </div>
</div>

<script>
(function() {
    const isTouchDevice = ('ontouchstart' in window) || (navigator.maxTouchPoints > 0);
    if (!isTouchDevice) return;

    const hudRoot = document.getElementById("mobileHudRoot");
    const modePill = document.getElementById("mobileModePill");
    const layerOffice = document.getElementById("layerOffice");
    const layerSalvage = document.getElementById("layerSalvage");
    const canvas = document.getElementById("canvas");

    hudRoot.style.display = "block";
    let mode = 0;
    const modes = [
        { label: "TOUCH: OFF", layer: null },
        { label: "TOUCH: MASK", layer: layerOffice },
        { label: "TOUCH: ROAM", layer: layerSalvage }
    ];

    const activeKeys = new Set();

    function setKey(codeStr, keyCode, down) {
        if (down && activeKeys.has(codeStr)) return;
        if (!down && !activeKeys.has(codeStr)) return;
        if (down) activeKeys.add(codeStr); else activeKeys.delete(codeStr);
        
        const keyMap = { KeyW: "w", KeyA: "a", KeyS: "s", KeyD: "d", Space: " ", ShiftLeft: "Shift" };
        const ev = new KeyboardEvent(down ? "keydown" : "keyup", {
            key: keyMap[codeStr] || "",
            code: codeStr,
            keyCode: keyCode,
            which: keyCode,
            bubbles: true,
            cancelable: true
        });
        canvas.focus();
        canvas.dispatchEvent(ev);
        window.dispatchEvent(ev);
        document.dispatchEvent(ev);
    }

    function releaseAll() {
        document.querySelectorAll(".t-btn").forEach(b => b.style.opacity = "0.35");
        [["KeyW",87],["KeyA",65],["KeyS",83],["KeyD",68],["Space",32],["ShiftLeft",16]].forEach(k => setKey(k[0], k[1], false));
    }

    modePill.addEventListener("touchstart", function (e) {
        e.preventDefault();
        e.stopPropagation();
        releaseAll();
        mode = (mode + 1) % modes.length;
        modePill.textContent = modes[mode].label;
        if (layerOffice) layerOffice.style.display = (modes[mode].layer === layerOffice) ? "block" : "none";
        if (layerSalvage) layerSalvage.style.display = (modes[mode].layer === layerSalvage) ? "block" : "none";
    }, { passive: false });

    document.querySelectorAll(".t-btn").forEach(btn => {
        const codeStr = btn.dataset.key;
        const keyCode = Number(btn.dataset.code);
        
        btn.addEventListener("touchstart", function (e) {
            e.preventDefault();
            e.stopPropagation();
            btn.style.opacity = "1";
            setKey(codeStr, keyCode, true);
        }, { passive: false });
        
        const end = function (e) {
            e.preventDefault();
            e.stopPropagation();
            btn.style.opacity = "0.35";
            setKey(codeStr, keyCode, false);
        };
        btn.addEventListener("touchend", end, { passive: false });
        btn.addEventListener("touchcancel", end, { passive: false });
    });

    const joyZone = document.getElementById("joyZone");
    const joyBase = document.getElementById("joyBase");
    const joyKnob = document.getElementById("joyKnob");
    let joyTouchId = null, originX = 0, originY = 0;

    joyZone.addEventListener("touchstart", function (e) {
        if (joyTouchId !== null) return;
        e.preventDefault();
        const t = e.changedTouches[0];
        joyTouchId = t.identifier;
        originX = t.clientX;
        originY = t.clientY;
        joyBase.style.left = originX + "px";
        joyBase.style.top = originY + "px";
        joyBase.style.display = "block";
        joyKnob.style.transform = "translate(0px, 0px)";
    }, { passive: false });

    joyZone.addEventListener("touchmove", function (e) {
        if (joyTouchId === null) return;
        e.preventDefault();
        for (let i = 0; i < e.changedTouches.length; i++) {
            const t = e.changedTouches[i];
            if (t.identifier !== joyTouchId) continue;
            const dx = t.clientX - originX;
            const dy = t.clientY - originY;
            const maxR = 34;
            const dist = Math.hypot(dx, dy);
            const clamp = dist > maxR ? maxR / dist : 1;
            
            joyKnob.style.transform = `translate(${dx * clamp}px, ${dy * clamp}px)`;
            
            const dead = 12;
            setKey("KeyW", 87, dy < -dead);
            setKey("KeyS", 83, dy > dead);
            setKey("KeyA", 65, dx < -dead);
            setKey("KeyD", 68, dx > dead);
        }
    }, { passive: false });

    function endJoy(e) {
        for (let i = 0; i < e.changedTouches.length; i++) {
            if (e.changedTouches[i].identifier === joyTouchId) {
                joyTouchId = null;
                joyBase.style.display = "none";
                setKey("KeyW", 87, false);
                setKey("KeyS", 83, false);
                setKey("KeyA", 65, false);
                setKey("KeyD", 68, false);
            }
        }
    }
    joyZone.addEventListener("touchend", endJoy, { passive: false });
    joyZone.addEventListener("touchcancel", endJoy, { passive: false });
})();
</script>
